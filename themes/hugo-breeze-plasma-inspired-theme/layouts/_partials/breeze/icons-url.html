{{- /* Builds one SVG sprite from every assets/breeze/icons/*.svg (theme + site; a site file with the same name replaces the theme's), fingerprints it, and returns its URL. Used through partialCached. */ -}} {{- $body := "" -}} {{- range resources.Match "breeze/icons/*.svg" -}} {{- $id := path.BaseName .Name -}} {{- $svg := .Content | replaceRE ` xmlns="[^"]*"` "" | strings.TrimSpace -}} {{- $svg = replaceRE `^$` "" $svg -}} {{- $body = print $body $svg -}} {{- end -}} {{- $sprite := resources.FromString "breeze/icons.svg" (print ``) | fingerprint -}} {{- return $sprite.RelPermalink -}}
